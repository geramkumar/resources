FILE: deployment/stage_volume_files.sh
======================================================================

#!/bin/bash

set -euo pipefail


# --------------------------------------------------------------------
# Usage
#
# ./deployment/stage_volume_files.sh dev
# ./deployment/stage_volume_files.sh preprod
# ./deployment/stage_volume_files.sh prod
# --------------------------------------------------------------------

TARGET="${1:?Usage: ./deployment/stage_volume_files.sh <dev|preprod|prod>}"


# --------------------------------------------------------------------
# Source Volume
#
# This is the Volume where the approved source files currently exist.
# --------------------------------------------------------------------

SOURCE_VOLUME="${SOURCE_SI_VOLUME:?Set SOURCE_SI_VOLUME first}"


# --------------------------------------------------------------------
# Determine target Volume
# --------------------------------------------------------------------

case "${TARGET}" in

  dev)

    CATALOG="catalog_dev"

    SCHEMA="service_intelligence_dev"

    ;;


  preprod)

    CATALOG="catalog_preprod"

    SCHEMA="service_intelligence_preprod"

    ;;


  prod)

    CATALOG="catalog_prod"

    SCHEMA="service_intelligence_prod"

    ;;


  *)

    echo "Invalid target: ${TARGET}"

    exit 1

    ;;

esac


TARGET_VOLUME="/Volumes/${CATALOG}/${SCHEMA}/si_volume"


echo "=================================================="

echo "Target      : ${TARGET}"

echo "Source      : ${SOURCE_VOLUME}"

echo "Target      : ${TARGET_VOLUME}"

echo "=================================================="


# --------------------------------------------------------------------
# Create directories
# --------------------------------------------------------------------

databricks fs mkdir \
  "${TARGET_VOLUME}/scripts"

databricks fs mkdir \
  "${TARGET_VOLUME}/libraries"


# --------------------------------------------------------------------
# Copy init script
# --------------------------------------------------------------------

echo "Copying init_script.sh..."

databricks fs cp \
  "${SOURCE_VOLUME}/scripts/init_script.sh" \
  "${TARGET_VOLUME}/scripts/init_script.sh" \
  --overwrite


# --------------------------------------------------------------------
# Copy requirements.txt
# --------------------------------------------------------------------

echo "Copying requirements.txt..."

databricks fs cp \
  "${SOURCE_VOLUME}/libraries/requirements.txt" \
  "${TARGET_VOLUME}/libraries/requirements.txt" \
  --overwrite


echo "Volume files staged successfully."

FILE: init_script.sh
======================================================================

#!/bin/bash

set -e


# ====================================================================
# 1. ENVIRONMENT VARIABLES
# ====================================================================

export SI_ENV1="RAM"

export SI_ENV2="SI"

export SI_ENV3="${SI_ENV3:-UNKNOWN}"


# ====================================================================
# 2. PYTHON PACKAGE INSTALLATION
# ====================================================================

REQUIREMENTS_FILE="/Volumes/${SI_CATALOG}/${SI_SCHEMA}/${SI_VOLUME}/libraries/requirements.txt"


echo "Environment  : ${SI_ENV3}"

echo "Installing packages from:"
echo "${REQUIREMENTS_FILE}"


if [ -f "${REQUIREMENTS_FILE}" ]; then

    /databricks/python/bin/pip install \
        -r "${REQUIREMENTS_FILE}"

else

    echo "ERROR: requirements.txt not found:"
    echo "${REQUIREMENTS_FILE}"

    exit 1

fi


echo "Init script completed successfully."

# Customer Intelligence - All Purpose Cluster

This DAB creates:

1. Unity Catalog Volume
2. Required Volume directories
3. init_script.sh in the target Volume
4. requirements.txt in the target Volume
5. All-Purpose Cluster
6. Cluster initialization using init_script.sh


----------------------------------------------------------------------
DEPLOYMENT SEQUENCE
----------------------------------------------------------------------

# 1. Validate bundle

databricks bundle validate -t dev


# 2. Deploy Volume

databricks bundle deploy -t dev


# 3. Stage files into Volume

export SOURCE_SI_VOLUME=/Volumes/<SOURCE-CATALOG>/<SOURCE-SCHEMA>/si_volume

./deployment/stage_volume_files.sh dev


# 4. Deploy/update cluster

databricks bundle deploy -t dev


----------------------------------------------------------------------
PREPROD
----------------------------------------------------------------------

databricks bundle validate -t preprod

databricks bundle deploy -t preprod

export SOURCE_SI_VOLUME=/Volumes/<SOURCE-CATALOG>/<SOURCE-SCHEMA>/si_volume

./deployment/stage_volume_files.sh preprod

databricks bundle deploy -t preprod


----------------------------------------------------------------------
PROD

databricks bundle validate -t prod

databricks bundle deploy -t prod

export SOURCE_SI_VOLUME=/Volumes/<SOURCE-CATALOG>/<SOURCE-SCHEMA>/si_volume

./deployment/stage_volume_files.sh prod

databricks bundle deploy -t prod
