#!/bin/bash

set -euo pipefail

TARGET="${1:?Target is required}"

echo "Staging SI volume files for target: ${TARGET}"

case "${TARGET}" in

  dev)
    CATALOG="catalog_dev"
    SCHEMA="service_intelligence_dev"
    ;;

  preprod)
    CATALOG="catalog_preprod"
    SCHEMA="service_intelligence_preprod"
    ;;

  prod)
    CATALOG="catalog_prod"
    SCHEMA="service_intelligence_prod"
    ;;

  *)
    echo "Invalid target: ${TARGET}"
    exit 1
    ;;

esac


TARGET_VOLUME="/Volumes/${CATALOG}/${SCHEMA}/si_volume"

SOURCE_VOLUME="${SOURCE_SI_VOLUME:?SOURCE_SI_VOLUME must be set}"


echo "Target volume: ${TARGET_VOLUME}"

echo "Creating directories..."

databricks fs mkdir "${TARGET_VOLUME}/scripts"

databricks fs mkdir "${TARGET_VOLUME}/drivers"

databricks fs mkdir "${TARGET_VOLUME}/libraries"


echo "Copying init script..."

databricks fs cp \
  "${SOURCE_VOLUME}/scripts/init_script.sh" \
  "${TARGET_VOLUME}/scripts/init_script.sh" \
  --overwrite


echo "Copying JDBC drivers..."

databricks fs cp \
  "${SOURCE_VOLUME}/drivers/" \
  "${TARGET_VOLUME}/drivers/" \
  --recursive \
  --overwrite


echo "Copying requirements.txt..."

databricks fs cp \
  "${SOURCE_VOLUME}/libraries/requirements.txt" \
  "${TARGET_VOLUME}/libraries/requirements.txt" \
  --overwrite


echo "Volume staging completed."
