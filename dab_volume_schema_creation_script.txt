# stage_volume_files.sh

#!/bin/bash
set -euo pipefail

TARGET_VOLUME="/Volumes/${CATALOG_NAME}/${SCHEMA_NAME}/si_volume"

echo "Staging files to ${TARGET_VOLUME}"

databricks fs cp \
  "volume_files/scripts/init_script.sh" \
  "${TARGET_VOLUME}/scripts/init_script.sh" \
  --overwrite

databricks fs cp \
  "volume_files/libraries/requirements.txt" \
  "${TARGET_VOLUME}/libraries/requirements.txt" \
  --overwrite

for jar in volume_files/jdbc/*.jar; do
    [ -e "$jar" ] || continue

    databricks fs cp \
      "$jar" \
      "${TARGET_VOLUME}/jdbc/$(basename "$jar")" \
      --overwrite
done

echo "Volume files staged successfully."
