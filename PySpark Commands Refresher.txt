spark.sparkContext.getConf().getAll()

🔹 Driver
spark.driver.memory 	- Driver heap for planning, metadata, broadcast, collect; best practice 8–16 GB in production pipelines.
spark.driver.cores		- Driver CPU cores for scheduling and planning; best practice 2–4 cores.
spark.driver.maxResultSize- Maximum data driver can fetch; best practice keep default 1 GB to prevent driver crashes.

🔹 Executor
spark.executor.memory 	- Executor JVM heap for execution and caching; best practice 8–16 GB to avoid OOM and GC pauses.
spark.executor.cores	- Parallel tasks per executor; best practice 3–5 cores to balance CPU utilization and garbage collection.
spark.executor.instances- Total executor JVMs; best practice dynamic allocation enabled, avoid fixed sizing in production workloads.
spark.executor.memoryOverhead - Off-heap memory for Python, shuffle, JVM; best practice 15–25% for PySpark workloads.


🔹 Memory Management
spark.memory.fraction	- Executor heap fraction for execution and storage; best practice keep default 0.6.
spark.memory.storageFraction- Storage share of Spark memory; best practice 0.5, reduce for shuffle-heavy workloads.

🔹 Shuffle & Parallelism
spark.sql.shuffle.partitions - Partitions after shuffle operations; best practice 2–4× total executor cores.

🔹 Serialization & Python
spark.serializer 	- Serialization mechanism; best practice KryoSerializer for lower memory footprint and faster shuffles.

from pyspark.sql import SparkSession
from pyspark.sql.window import Window
from pyspark.sql.functions import (
    col, row_number, rank, dense_rank,
    lag, lead, sum, avg, max, min
)

spark = SparkSession.builder.appName("WindowFunctions").getOrCreate()

# Sample DataFrame
df = spark.createDataFrame(
    [
        (1, "IT", 50000),
        (2, "IT", 60000),
        (3, "HR", 40000),
        (4, "HR", 45000),
        (5, "HR", 45000)
    ],
    ["emp_id", "dept", "salary"]
)

dept_order_window = Window.partitionBy("dept").orderBy(col("salary").desc())   # Partition + ordering
dept_window = Window.partitionBy("dept")                                       # Partition only

# Ranking Functions
df = (
    df
    .withColumn("row_number", row_number().over(dept_order_window))  # Unique sequential number per partition
    .withColumn("rank", rank().over(dept_order_window))              # Same rank for ties, with gaps
    .withColumn("dense_rank", dense_rank().over(dept_order_window))  # Same rank for ties, no gaps
)

# Lead & Lag Functions
df = (
    df
    .withColumn("prev_salary", lag("salary", 1).over(dept_order_window))   # Previous row value
    .withColumn("next_salary", lead("salary", 1).over(dept_order_window))  # Next row value
)

# Aggregate Window Functions
df = (
    df
    .withColumn("dept_total_salary", sum("salary").over(dept_window))  # Sum per department
    .withColumn("dept_avg_salary", avg("salary").over(dept_window))    # Average per department
    .withColumn("dept_max_salary", max("salary").over(dept_window))    # Max per department
    .withColumn("dept_min_salary", min("salary").over(dept_window))    # Min per department
)

df.show(truncate=False)

# Interview Notes:
# - row_number() is used for deterministic deduplication
# - rank() creates gaps when ties exist
# - dense_rank() avoids gaps
# - Window functions do not reduce rows
# - Overuse of window functions can increase shuffle cost

from pyspark.sql import SparkSession
from pyspark import StorageLevel

spark = SparkSession.builder.appName("PersistLevels").getOrCreate()

df.persist(StorageLevel.MEMORY_ONLY)        # Stores deserialized data in memory; recomputes if memory is insufficient
df.persist(StorageLevel.MEMORY_ONLY_SER)    # Stores serialized data in memory; saves memory but adds CPU overhead
df.persist(StorageLevel.MEMORY_AND_DISK)    # Stores in memory and spills to disk; default used by cache()
df.persist(StorageLevel.MEMORY_AND_DISK_SER)# Serialized memory + disk spill; efficient for large datasets
df.persist(StorageLevel.DISK_ONLY)          # Stores only on disk; slower but avoids recomputation
df.persist(StorageLevel.OFF_HEAP)            # Stores data outside JVM heap; reduces GC overhead (needs off-heap enabled)

# cache() == persist(StorageLevel.MEMORY_AND_DISK)

from pyspark.sql import SparkSession
from pyspark.sql.functions import col, row_number, when, desc
from pyspark.sql.window import Window

data = [
    (1, "Ram", 10000, "IT", "2024-01-01"),
    (1, "Ram", None,  "IT", "2024-01-10"),  # duplicate emp_id, newer record
    (2, "Sam", None,  None, "2024-01-05"),  # null salary & dept
    (3, None,  15000, "HR", "2024-01-03"),  # null name
    (3, None,  15000, "HR", "2024-01-03"),  # exact duplicate row
    (None, "John", 12000, "FIN", "2024-01-02")  # null business key
]

columns = ["emp_id", "name", "salary", "dept", "update_ts"]

df = spark.createDataFrame(data, columns)
df_valid = df.dropna(subset=["emp_id"])

# --------------------------------------------
# 4. Deterministic Deduplication
# Keep latest record per emp_id using window
# --------------------------------------------
window_spec = Window.partitionBy("emp_id").orderBy(desc("update_ts"))

df_dedup = (
    df_valid
    .withColumn("rn", row_number().over(window_spec))
    .filter(col("rn") == 1)
    .drop("rn")
)

# 5. Remove Exact Duplicate Rows (Safety Net)
df_dedup = df_dedup.dropDuplicates()

# 6. Handle NULL Values (Imputation)

# Fill simple NULLs
df_filled = df_dedup.fillna({
    "salary": 0,
    "dept": "UNKNOWN",
    "name": "NA"
})

# 7. Conditional NULL Handling (when / otherwise)
df_final = df_filled.withColumn(
    "salary",
    when(col("salary").isNull() | (col("salary") < 0), 0)
    .otherwise(col("salary"))
)

from pyspark.sql import SparkSession

# 1. Spark Session + Configurations

    .appName("SeniorPySparkSQLRefresher") \
    .config("spark.sql.shuffle.partitions", "4") \   # Controls shuffle parallelism
    .config("spark.sql.adaptive.enabled", "true") \  # Enables AQE
    .enableHiveSupport() \                            # Enables Hive metastore & SQL tables
    .getOrCreate()

# 2. Create Database (SQL)
spark.sql("CREATE DATABASE IF NOT EXISTS interview_db")   # Creates database
spark.sql("USE interview_db")                              # Switches database

# 3. Create DataFrames (Source Data)
emp_df = spark.createDataFrame(
    [(1, "Ram", "IT", 50000),
     (2, "Sam", "IT", 60000),
     (3, "John", "HR", 40000)],
    ["emp_id", "name", "dept", "salary"]
)

dept_df = spark.createDataFrame(
    [("IT", "Technology"),
     ("HR", "Human Resources")],
    ["dept", "dept_name"]
)

# 4. Create Temporary Views (DF → SQL)

emp_df.createOrReplaceTempView("emp")      # Session-scoped temp view
dept_df.createOrReplaceTempView("dept")    # Enables SQL queries on DF

# 5. Basic SQL Queries

spark.sql("SELECT * FROM emp").show()      # Select query
spark.sql("SELECT emp_id, salary FROM emp WHERE salary > 45000").show()  # Filter


# """ is for multiline queries
# 6. Join Using SQL

joined_df = spark.sql("""
    SELECT e.emp_id, e.name, d.dept_name, e.salary
    FROM emp e
    JOIN dept d
    ON e.dept = d.dept
""")                                        # SQL join between views

joined_df.show()

# 7. Aggregation (SQL)

agg_df = spark.sql("""
    SELECT dept_name, SUM(salary) AS total_salary, COUNT(*) AS emp_count
    FROM (
        SELECT e.salary, d.dept_name
        FROM emp e JOIN dept d ON e.dept = d.dept
    )
    GROUP BY dept_name
""")                                        # GROUP BY aggregation

agg_df.show()

# 8. SQL Result → DataFrame Operations

agg_df.filter(col("total_salary") > 50000).show()   # DataFrame API on SQL result

# 9. Create Managed Table from DataFrame

agg_df.write.mode("overwrite").saveAsTable("dept_salary_summary")  # Stores as Hive table

# 10. Query Managed Table (SQL)

spark.sql("SELECT * FROM dept_salary_summary").show()  # Reads from metastore table

emp_df.write.mode("overwrite").format("parquet").saveAsTable("emp_table")  # Writes DF as table

# 13. SQL Table → DataFrame

emp_table_df = spark.table("emp_table")                 # Reads table as DataFrame

# 14. Drop Table & Database (Cleanup)

spark.sql("DROP TABLE IF EXISTS dept_salary_summary")   # Drops table
