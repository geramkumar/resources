```bash
# ============================================================
# 🔷 DOCKER COMPLETE FLOW SCRIPT WITH DETAILED COMMENTS
# ============================================================

# ---------- 1. VERIFY DOCKER INSTALLATION AND ENVIRONMENT ----------

docker --version
# This command checks installed Docker client version to confirm installation is successful

docker info
# This command shows detailed Docker engine configuration like storage driver, memory, and runtime

# ---------- 2. IMAGE MANAGEMENT COMMANDS FOR DOWNLOADING AND VIEWING ----------

docker pull nginx
# This command downloads the latest nginx image from Docker Hub registry for running web servers

docker pull postgres:15
# This command pulls a specific version of postgres image useful for consistent database environments

docker images
# This command lists all locally available Docker images including repository, tag, size, and image id

docker image ls
# This command provides the same output as docker images but using newer standardized syntax

docker rmi nginx
# This command removes nginx image from local system if no containers are dependent on it

docker rmi -f nginx
# This command forcefully removes image even if containers are referencing it currently

# ---------- 3. CONTAINER MANAGEMENT COMMANDS FOR RUNNING APPLICATIONS ----------

docker run nginx
# This command runs nginx container in foreground mode which blocks terminal until stopped manually

docker run -d -p 8080:80 --name my-nginx nginx
# This command runs nginx in background mode with port mapping and assigns a friendly container name

docker ps
# This command lists all currently running containers showing id, image, ports, and container status

docker ps -a
# This command lists all containers including stopped ones useful for debugging previous executions

docker inspect my-nginx
# This command provides deep JSON metadata including IP address, mounts, and configuration settings

docker logs my-nginx
# This command displays application logs generated by container useful for troubleshooting runtime issues

docker logs -f my-nginx
# This command streams logs in real-time similar to tail command for continuous monitoring purposes

docker exec -it my-nginx bash
# This command opens interactive terminal session inside container for debugging or manual inspection

docker exec -it my-nginx sh
# This command provides alternative shell when bash is not available in minimal container images

docker top my-nginx
# This command shows running processes inside container similar to Linux top command output

docker stats
# This command displays real-time CPU, memory, and network usage of all running containers

docker stop my-nginx
# This command gracefully stops running container allowing application to shutdown cleanly

docker start my-nginx
# This command starts previously stopped container without recreating it from scratch

docker restart my-nginx
# This command stops and immediately starts container useful after configuration or code changes

docker cp file.txt my-nginx:/app/
# This command copies local file into container filesystem useful for quick debugging or patching

docker rm my-nginx
# This command removes stopped container permanently freeing associated container resources

docker rm -f my-nginx
# This command forcefully removes running container without requiring manual stop operation

# ---------- 4. BUILD CUSTOM IMAGE USING DOCKERFILE ----------

# Sample Dockerfile (keep in same directory before running build command)
# FROM python:3.11
# WORKDIR /app
# COPY . .
# RUN pip install flask
# CMD ["python", "app.py"]

docker build -t my-python-app .
# This command builds custom image using Dockerfile instructions present in current directory context

docker tag my-python-app myrepo/my-python-app:v1
# This command tags local image with repository name to prepare for pushing to Docker Hub

docker run -d -p 5000:5000 my-python-app
# This command runs custom python application image exposing application on port 5000 externally

# ---------- 5. DOCKER COMPOSE REAL-WORLD SCENARIO ----------

# Create docker-compose.yml file before executing below commands
# This example runs a web application connected to postgres database using service networking

# version: "3.9"
# services:
#   web:
#     image: nginx
#     ports:
#       - "8080:80"
#     depends_on:
#       - db
#
#   db:
#     image: postgres:15
#     environment:
#       POSTGRES_USER: admin
#       POSTGRES_PASSWORD: secret
#       POSTGRES_DB: appdb
#     volumes:
#       - pgdata:/var/lib/postgresql/data
#
# volumes:
#   pgdata:

docker-compose up
# This command starts all defined services in foreground showing logs for web and database containers

docker-compose up -d
# This command starts services in detached mode allowing terminal reuse while containers run in background

docker-compose ps
# This command lists status of all services defined in compose file including ports and health state

docker-compose logs
# This command aggregates logs from all services useful for understanding multi-container interactions

docker-compose logs -f
# This command streams logs continuously helping identify issues between web and database communication

docker-compose up --build
# This command rebuilds images if Dockerfile changed and then starts updated services accordingly

docker-compose down
# This command stops and removes all containers, networks, and default resources created by compose

# ---------- 6. VOLUMES FOR DATA PERSISTENCE ACROSS CONTAINER RESTARTS ----------

docker volume create my-volume
# This command creates persistent storage volume which survives container deletion and recreation cycles

docker volume ls
# This command lists all volumes available locally helping identify unused or active storage resources

docker run -d -v my-volume:/data nginx
# This command mounts volume inside container ensuring data written persists beyond container lifecycle

docker volume rm my-volume
# This command removes volume permanently so ensure no important data exists before execution

# ---------- 7. NETWORKING FOR CONTAINER COMMUNICATION ----------

docker network ls
# This command lists all networks including bridge, host, and custom networks created by user

docker network create my-network
# This command creates isolated network allowing secure communication between related containers

docker run -d --network my-network nginx
# This command runs container attached to custom network enabling communication with other containers

# ---------- 8. CLEANUP COMMANDS TO FREE SYSTEM RESOURCES ----------

docker container prune
# This command removes all stopped containers to free disk space and avoid unnecessary clutter

docker image prune
# This command removes dangling images which are unused and not tagged by any repository

docker system prune
# This command removes unused containers, networks, and dangling images after confirmation prompt

docker system prune -a
# This command aggressively removes all unused images including tagged ones not used by containers

# ---------- 9. DOCKER HUB AUTHENTICATION AND IMAGE PUSH ----------

docker login
# This command authenticates user with Docker Hub registry to allow pushing private or public images

docker push myrepo/my-python-app:v1
# This command uploads tagged image to Docker Hub repository for sharing or deployment usage

docker logout
# This command logs out from Docker registry ensuring credentials are not stored locally anymore

# ============================================================
# 🔥 END OF COMPLETE DOCKER FLOW SCRIPT
# ============================================================
```
